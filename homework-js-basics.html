في الحصة تعلمنا كيف يتواصل JavaScript مع صفحة HTML — نختار عنصراً، ثم نغيّر مظهره أو محتواه.

script.js — كود الحصة 
// تحديد عناصر من الصفحة
const h1 = document.querySelector('h1');
h1.style.color = 'blue';

const h2 = document.querySelector('h2');
h2.style.color = 'purple';

// متغيرات وعمليات حسابية
let x = 5.5;
let y = 9;
let z = x / y;

// متغيرات منطقية (boolean)
let isOn  = true;
let isOff = false;

// شرط if — يغيّر النص إذا كان isOn يساوي true
if (isOn) {
    h2.textContent = `${z}`;
}
💡 querySelector تبحث عن أوّل عنصر يطابق المحدّد (مثل CSS).
    getElementById تبحث بالـ id مباشرةً وهي أسرع.

نشاط ٢ — العداد التلقائي
setInterval · getElementById · if / else if / else

في الحصة رأينا كيف يُحدّث رصد درجة الحرارة كل ثانية تلقائياً. الكود أسفله هو نفس الكود الذي شاهدناه معاً — اقرأه أولاً وافهم كل سطر.

نموذج رصد — script.js 
// نختار العنصر بالـ id
const myTemp = document.getElementById("temp");

// setInterval تكرّر الكود كل 1000 ميلي ثانية (= 1 ثانية)
setInterval(() => {
    let temp = Math.floor(Math.random() * 50);
    let text = "";

    if (temp > 30) {
        text = "حار 🔥";
    } else if (temp > 20) {
        text = "معتدل ☀️";
    } else {
        text = "بارد ❄️";
    }

    myTemp.textContent = temp + " °C — " + text;
}, 1000);

تحدّيات التعديل:
✏️ عدّل الكود وراقب النتيجة:
تحدي ①
غيّر 1000 إلى 3000 — بكم ثانية يتغيّر الرقم الآن؟
تحدي ②
أضف شرطاً رابعاً: إذا كانت الحرارة أقل من 10 درجات، النص يصبح بارد جداً 🥶
(تلميح: ضعه قبل الـ else الأخير)
تحدي ③ — إضافي 🌟
أضف عنصراً ثانياً في HTML بـ id="humidity" وأنشئ عداداً منفصلاً للرطوبة (0–100) داخل نفس setInterval.
✅ لا يوجد إجابة خاطئة — المهم أنك جرّبت وعرفت ماذا يفعل كل سطر.
